states = ['idle','busy']
actions = ['pick','move','charge']

R = {'idle':{'pick':5,'move':2,'charge':1},
     'busy':{'pick':3,'move':4,'charge':2}}

policy = {s:actions[0] for s in states}
V = {s:0 for s in states}
g = 0.9

for _ in range(5):
    # evaluation
    for _ in range(5):
        for s in states:
            V[s] = R[s][policy[s]] + g*V[s]
    # improvement
    for s in states:
        policy[s] = max(actions, key=lambda a:R[s][a]+g*V[s])

print("Values:", V)
print("Policy:", policy)

 output:

Values: {'idle': 46.41051006154074, 'busy': 36.63053949101875}
Policy: {'idle': 'pick', 'busy': 'move'}
