states = ['low','med','high']
actions = ['short','long']

P = {s:{a:[(1.0,s)] for a in actions} for s in states}
R = {'low':{'short':5,'long':3},
     'med':{'short':3,'long':4},
     'high':{'short':1,'long':5}}

V = {s:0 for s in states}
g = 0.9

for _ in range(10):
    for s in states:
        V[s] = max(R[s][a] + g*V[s] for a in actions)

policy = {s:max(actions, key=lambda a:R[s][a]+g*V[s]) for s in states}

print("Values:", V)
print("Policy:", policy)
