# 2-player Tic-Tac-Toe

board = [" "]*9

def print_b():
    for i in range(3):
        print("|".join(board[i*3:(i+1)*3]))
        if i<2: print("-"*5)

def winner():
    lines = [(0,1,2),(3,4,5),(6,7,8),
             (0,3,6),(1,4,7),(2,5,8),
             (0,4,8),(2,4,6)]
    for a,b,c in lines:
        if board[a]==board[b]==board[c]!=" ":
            return board[a]
    return None

print("Tic Tac Toe: Player 1 = X, Player 2 = O")
turn = "X"

while True:
    print_b()
    try:
        n = int(input(f"Player {turn}, enter position (1-9): ")) - 1
        if board[n] != " ":
            print("Invalid! Try again.")
            continue
        board[n] = turn
    except:
        print("Invalid! Try again.")
        continue

    w = winner()
    if w:
        print_b()
        print(f"Player {w} wins!")
        break
    if " " not in board:
        print_b()
        print("Draw!")
        break

    turn = "O" if turn=="X" else "X"
