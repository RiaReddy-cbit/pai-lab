def backtrack(a,v,d):
    if len(a)==len(v): return a
    x=[i for i in v if i not in a][0]
    for val in d[x]:
        if all(a.get(n)!=val for n in adj[x]):
            a[x]=val
            r=backtrack(a,v,d)
            if r:return r
            del a[x]

n=int(input("Enter number of regions: "))
vars=[]
adj={}

for i in range(n):
    node=input("Region name: ")
    vars.append(node)
    adj[node]=input("Neighbours (space separated): ").split()

colors=input("Enter colors: ").split()
domains={i:colors for i in vars}

res=backtrack({},vars,domains)

print("Solution found:")
for k in res:
    print(f"{k}: {res[k]}")
