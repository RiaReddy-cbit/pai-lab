import numpy as np

states = ['Interested','Neutral','Disengaged']
observations = ['Clicks','Low_Time']

start = np.array([0.5,0.3,0.2])
trans = np.array([[0.6,0.3,0.1],
                  [0.3,0.4,0.3],
                  [0.2,0.3,0.5]])
emit = np.array([[0.7,0.3],
                 [0.5,0.5],
                 [0.2,0.8]])

obs_seq = [0,1,1]  # Clicks, Low_Time, Low_Time

V = [{}]
path = {}

for i in range(len(states)):
    V[0][i] = start[i] * emit[i][obs_seq[0]]
    path[i] = [states[i]]

for t in range(1,len(obs_seq)):
    V.append({})
    new_path = {}
    for j in range(len(states)):
        prob, state = max((V[t-1][i]*trans[i][j]*emit[j][obs_seq[t]], i)
                          for i in range(len(states)))
        V[t][j] = prob
        new_path[j] = path[state] + [states[j]]
    path = new_path

prob, state = max((V[-1][i], i) for i in range(len(states)))

print("Most likely user engagement state sequence is:", path[state])
print("Probability of most likely state sequence is", prob)

OUTPUT:

Most likely user engagement state sequence is: ['Neutral', 'Disengaged', 'Disengaged']
Probability of most likely state sequence is 0.0144
